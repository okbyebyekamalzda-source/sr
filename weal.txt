		function cheat:get_target()
			local current_target = nil
			local maximum_distance = flags["combat maximum distance"]

			if flags["combat target"] and table.find(flags["combat target"], "players") then
				for i, v in pairs(game.Players:GetPlayers()) do
					if v == cheat.player then
						continue
					end
					if v.Character and v.Character:FindFirstChild("HumanoidRootPart") then
						local position, on_screen =
							cheat.camera:WorldToViewportPoint(v.Character.HumanoidRootPart.Position)
						if not on_screen then
							continue
						end
						local distance

						if flags["combat target type"] == "closest to mouse" then
							distance = (
								Vector2.new(position.X, position.Y - game.GuiService:GetGuiInset().Y)
								- Vector2.new(cheat.mouse.X, cheat.mouse.Y)
							).Magnitude
						elseif flags["combat target type"] == "closest to screen center" then
							local viewport_size = cheat.camera.ViewportSize
							local screen_center = Vector2.new(viewport_size.X / 2, viewport_size.Y / 2)
							distance = (Vector2.new(position.X, position.Y) - screen_center).Magnitude
						end

						local visible = cheat:visible_check(v.Character:FindFirstChild("Head"))
						local humanoid = v.Character:FindFirstChild("Humanoid")

						-- // fov check
						if flags["drawings fov"] and distance > flags["drawings fov radius"] then
							continue
						end

						-- // distance check
						if distance > maximum_distance then
							continue
						end

						-- // health check
						if
							flags["combat conditions"]
							and table.find(flags["combat conditions"], "health")
							and humanoid
							and humanoid.Health <= flags["combat minimum health"]
						then
							continue
						end

						-- // visible check
						if
							flags["combat conditions"]
							and table.find(flags["combat conditions"], "visible")
							and not visible
						then
							continue
						end

						-- // invisible check
						if
							flags["combat conditions"]
							and table.find(flags["combat conditions"], "invisible")
							and v.Character:FindFirstChild("Head")
							and v.Character:FindFirstChild("Head").Transparency == 1
						then
							continue
						end

						-- // team check
						if
							flags["combat conditions"]
							and table.find(flags["combat conditions"], "team")
							and cheat.friendlies
							and table.find(cheat.friendlies, v)
						then
							continue
						end

						-- ￢ﾜﾅ Found a new target
						current_target = v
						maximum_distance = distance
					end
				end
			end
			if current_target ~= cheat.target then
				if current_target and flags["combat notify target"] then
					library:notification({
						text = "targeting " .. current_target.Name,
						time = 5,
					})
				end
				cheat.target = current_target
			end

			return current_target
		end

		function cheat:pass_through(from, target, params, ignoreTable)
			params.FilterDescendantsInstances = ignoreTable
			local Result = workspace:Raycast(from, (target.Position - from).unit * 10000, params)
			if Result then
				local Instance = Result.Instance
				if Instance:IsDescendantOf(target.Parent) then
					return true
				elseif Instance.CanCollide == false or Instance.Transparency == 1 then
					if Instance.Name ~= "Head" and Instance.Name ~= "HumanoidRootPart" then
						table.insert(ignoreTable, Instance)
						cheat:pass_through(Result.Position, target, params, ignoreTable)
					end

		function cheat:visible_check(target)
			if not cheat.player.Character or not cheat.player.Character:FindFirstChild("Head") then
				return false
			end

			local params = RaycastParams.new()
			params.FilterType = Enum.RaycastFilterType.Blacklist
			local ignoreTable = { cheat.camera, cheat.player.Character }
			params.FilterDescendantsInstances = ignoreTable
			params.IgnoreWater = true
			local From = cheat.camera.CFrame.p
			local Result = workspace:Raycast(From, (target.Position - From).unit * 10000, params)

			if Result then
				local Instance = Result.Instance
				if Instance:IsDescendantOf(target.Parent) then
					return true
				elseif Instance.CanCollide == false or Instance.Transparency == 1 then
					if Instance.Name ~= "Head" and Instance.Name ~= "HumanoidRootPart" then
						table.insert(ignoreTable, Instance)
						cheat:pass_through(Result.Position, target, params, ignoreTable)
					end
				end
			end
			return false
		end

		function cheat:ValidateRay(Args, RayMethod)
			local Matches = 0
			if #Args < RayMethod.ArgCountRequired then
				return false
			end
			for Pos, Argument in next, Args do
				if typeof(Argument) == RayMethod.Args[Pos] then
					Matches = Matches + 1
				end
			end
			return Matches >= RayMethod.ArgCountRequired
		end

		function cheat:update_hitmarkers()
			for Index, Data in pairs(cheat.hitmarkers) do 
				task.spawn(function()
					if not Data.Hitmarker then 
						local Hitmarker = Instance.new("ImageLabel", cheat.sgui)
						Hitmarker.Image = "http://www.roblox.com/asset/?id=13298929624"
						Hitmarker.BackgroundTransparency = 1
						Hitmarker.Size = UDim2.new(0, 50, 0, 50)
						Hitmarker.Visible = false
						Hitmarker.Rotation = 45
						Hitmarker.AnchorPoint = Vector2.new(0.5, 0.5)

						Data.Hitmarker = Hitmarker
					end
					local Hitmarker = Data.Hitmarker
					local Position, IsOnScreen = cheat.camera:WorldToViewportPoint(Data.Position)
					Hitmarker.Visible = IsOnScreen
					local Transparency = 0

					if tick() - Data.StartedAt > flags["hit marker life"] and not Data.IsFading then 
						Data.FadeTick = tick()
						Data.IsFading = true
					end

					if Data.IsFading then 
						local TimeDifference = tick() - Data.FadeTick
						Transparency = math.min(1, TimeDifference / flags["hit marker life"])
					end

					if Hitmarker.Visible then 
						Hitmarker.ImageTransparency = Transparency
						Hitmarker.Position = UDim2.new(0, Position.X, 0, Position.Y)
						Hitmarker.ImageColor3 = flags["hit marker color"].Color
					end

					if Hitmarker.ImageTransparency == 1 then
						cheat.hitmarkers[Index] = nil
						Hitmarker:Destroy()
					end
				end)
			end
		end

-- ===== INSERTED: BoxChams replacement for enemy ESP chams =====
-- This block replaces the enemy ESP chams visuals by using BoxHandleAdornment.
-- It watches the existing UI flag "chams" (the enemies toggle) and uses its state.

do
    -- safe references
    local Players = game:GetService("Players")
    local LocalPlayer = Players.LocalPlayer
    local RunService = game:GetService("RunService")

    -- helper: try to read UI flags table; fall back to getgenv values if absent
    local function is_chams_enabled()
        local ok, v = pcall(function() return flags and flags["chams"] end)
        if ok and v ~= nil then return v end
        -- fallback to global if UI flags not available yet
        if getgenv and getgenv().BoxChams_Enabled ~= nil then
            return getgenv().BoxChams_Enabled
        end
        return false
    end

    local function get_chams_color()
        -- check UI color pickers first
        local ok, col_visible = pcall(function() return flags and flags["chams_invisible_color"] and flags["chams_invisible_color"].Color end)
        if ok and col_visible then return col_visible end
        if getgenv and getgenv().BoxChams_Color then return getgenv().BoxChams_Color end
        return Color3.fromRGB(255,150,150)
    end

    local function get_chams_transparency()
        local ok, t = pcall(function() return flags and flags["chams_transparency"] end)
        if ok and t then return t end
        if getgenv and getgenv().BoxChams_Transparency then return getgenv().BoxChams_Transparency end
        return 0.4
    end

    -- remove previous adornments created by this module
    local function clearBoxChams(character)
        if not character then return end
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BoxHandleAdornment") and part.Name == "ChamBox" then
                part:Destroy()
            end
            -- also remove named attachments left by prior runs
            if part:IsA("Instance") and part.Name == "ChamBox" and not part:IsA("BoxHandleAdornment") then
                pcall(function() part:Destroy() end)
            end
        end
    end

    -- apply box chams to a character's baseparts
    local function applyBoxChams(character)
        if not character then return end
        clearBoxChams(character)
        for _, limb in ipairs(character:GetChildren()) do
            if limb:IsA("BasePart") then
                local cham = Instance.new("BoxHandleAdornment")
                cham.Name = "ChamBox"
                cham.Adornee = limb
                cham.Size = limb.Size
                cham.ZIndex = 10
                cham.AlwaysOnTop = true
                cham.Color3 = get_chams_color()
                cham.Transparency = get_chams_transparency()
                -- ensure adornment is parented to PlayerGui or coregui to render
                cham.Parent = game:GetService("CoreGui")
                -- store reference on limb for cleanup convenience
                pcall(function() 
                    local tag = Instance.new("BoolValue")
                    tag.Name = "ChamBoxTag"
                    tag.Parent = limb
                    tag.Value = true
                end)
            end
        end
    end

    -- main loop: apply to all enemy players when flags["chams"] is true
    task.spawn(function()
        while task.wait(0.25) do
            local enabled = is_chams_enabled()
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer then
                    local character = plr.Character
                    if character then
                        if enabled then
                            -- try to apply only to enemies if team logic exists
                            local is_enemy = true
                            pcall(function()
                                if flags and flags["esp_team_mode"] then
                                    -- not assuming exact team logic; keep default of applying to others
                                end
                            end)
                            if is_enemy then
                                pcall(function() applyBoxChams(character) end)
                            end
                        else
                            pcall(function() clearBoxChams(character) end)
                        end
                    end
                end
            end
        end
    end)

    -- reapply/cleanup on respawn
    Players.PlayerAdded:Connect(function(plr)
        plr.CharacterAdded:Connect(function(char)
            task.wait(0.5)
            if is_chams_enabled() then
                pcall(function() applyBoxChams(char) end)
            end
        end)
    end)
end
